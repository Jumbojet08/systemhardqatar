# Registry path for PowerShell Transcription setting
$regPath = "HKLM:\Software\Policies\Microsoft\Windows\PowerShell"

# Check if the registry path exists; if not, create it
if (-not (Test-Path $regPath)) {
    Write-Host "Registry path not found. Creating: $regPath" -ForegroundColor Yellow
    New-Item -Path $regPath -Force | Out-Null
}

# Set the registry value to disable PowerShell Transcription
Set-ItemProperty -Path $regPath -Name "EnableTranscripting" -Value 0

Write-Host "PowerShell Transcription has been disabled." -ForegroundColor Green
