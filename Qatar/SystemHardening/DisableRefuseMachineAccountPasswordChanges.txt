# Path for "DisablePasswordChange" setting
$secpolicy = "HKLM:\System\CurrentControlSet\Services\Netlogon\Parameters"

# Check if the registry path exists; if not, create it
if (-not (Test-Path $secpolicy)) {
    Write-Host "Registry path not found. Creating: $secpolicy" -ForegroundColor Yellow
    New-Item -Path $secpolicy -Force | Out-Null
}

# Set the registry value to disable machine account password changes
Set-ItemProperty -Path $secpolicy -Name "DisablePasswordChange" -Value 1

Write-Host "Machine account password changes have been disabled." -ForegroundColor Green
