# Registry path for "DoNotUseTemporaryFoldersPerSession" setting
$regPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\Advanced"

# Check if the registry path exists; if not, create it
if (-not (Test-Path $regPath)) {
    Write-Host "Registry path not found. Creating: $regPath" -ForegroundColor Yellow
    New-Item -Path $regPath -Force | Out-Null
}

# Set the registry value to enable "Do not use temporary folders per session"
Set-ItemProperty -Path $regPath -Name "DoNotUseTemporaryFoldersPerSession" -Value 1

Write-Host "Temporary folders will not be used per session." -ForegroundColor Green
