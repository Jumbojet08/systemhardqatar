# Registry path for PIN sign-in settings
$regPath = "HKLM:\Software\Policies\Microsoft\Windows\System"

# Check if the registry path exists; if not, create it
if (-not (Test-Path $regPath)) {
    Write-Host "Registry path not found. Creating: $regPath" -ForegroundColor Yellow
    New-Item -Path $regPath -Force | Out-Null
}

# Set the registry value to disable convenience PIN sign-in
Set-ItemProperty -Path $regPath -Name "AllowDomainPINLogon" -Value 0

Write-Host "Convenience PIN sign-in has been disabled." -ForegroundColor Green
