# Registry path to disable suggested apps in Windows Ink Workspace
$regPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Policies\Explorer"

# Check if the registry path exists; if not, create it
if (-not (Test-Path $regPath)) {
    Write-Host "Registry path not found. Creating: $regPath" -ForegroundColor Yellow
    New-Item -Path $regPath -Force | Out-Null
}

# Set the registry value to disable suggested apps in Windows Ink Workspace
Set-ItemProperty -Path $regPath -Name "DisableSuggestedApps" -Value 1

Write-Host "Suggested apps in Windows Ink Workspace have been disabled." -ForegroundColor Green
