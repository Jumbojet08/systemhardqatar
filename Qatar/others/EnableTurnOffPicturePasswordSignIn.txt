# Registry path for picture password sign-in settings
$regPath = "HKLM:\Software\Policies\Microsoft\Windows\System"

# Check if the registry path exists; if not, create it
if (-not (Test-Path $regPath)) {
    Write-Host "Registry path not found. Creating: $regPath" -ForegroundColor Yellow
    New-Item -Path $regPath -Force | Out-Null
}

# Set the registry value to disable picture password sign-in
Set-ItemProperty -Path $regPath -Name "BlockUserPassword" -Value 1

Write-Host "Picture password sign-in has been turned off." -ForegroundColor Green
