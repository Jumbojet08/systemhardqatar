# Registry path for Windows Ink Workspace settings
$regPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Policies\Explorer"

# Check if the registry path exists; if not, create it
if (-not (Test-Path $regPath)) {
    Write-Host "Registry path not found. Creating: $regPath" -ForegroundColor Yellow
    New-Item -Path $regPath -Force | Out-Null
}

# Set the registry value to disable Windows Ink Workspace
Set-ItemProperty -Path $regPath -Name "DisableWindowsInkWorkspace" -Value 1

Write-Host "Windows Ink Workspace has been disabled." -ForegroundColor Green
