function EnablePointAndPrintRestrictionsWhenUpdatingDrivers {
    $RegPath = "HKLM:\Software\Policies\Microsoft\Windows NT\Printers\PointAndPrint"

    # Ensure the registry path exists
    if (!(Test-Path $RegPath)) {
        New-Item -Path $RegPath -Force | Out-Null
    }

    # Set restrictions when updating drivers
    Set-ItemProperty -Path $RegPath -Name "NoWarningNoElevationOnUpdate" -Value 0 -Type DWord -Force
    Set-ItemProperty -Path $RegPath -Name "UpdatePromptSettings" -Value 0 -Type DWord -Force

    Write-Host "Point and Print restrictions enabled for updating printer drivers." -ForegroundColor Green
}

# Call the function
EnablePointAndPrintRestrictionsWhenUpdatingDrivers
