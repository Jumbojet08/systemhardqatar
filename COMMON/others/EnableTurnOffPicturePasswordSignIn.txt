function EnableTurnOffPicturePasswordSignIn {
    $RegPath = "HKLM:\Software\Policies\Microsoft\Windows\System"

    # Ensure the registry path exists
    if (!(Test-Path $RegPath)) {
        New-Item -Path $RegPath -Force | Out-Null
    }

    # Turn off picture password sign-in
    Set-ItemProperty -Path $RegPath -Name "BlockPicturePassword" -Value 1 -Type DWord -Force

    Write-Host "Picture password sign-in has been disabled." -ForegroundColor Green
}

# Call the function
EnableTurnOffPicturePasswordSignIn
