function DisableAllowSuggestedAppsInWindowsInkWorkspace {
    $regPath = "HKLM:\Software\Policies\Microsoft\WindowsInkWorkspace"

    # Ensure the registry path exists
    if (!(Test-Path $regPath)) {
        New-Item -Path $regPath -Force | Out-Null
    }

    # Disable Suggested Apps in Windows Ink Workspace (0 = Disabled)
    Set-ItemProperty -Path $regPath -Name "AllowSuggestedAppsInWindowsInkWorkspace" -Value 0 -Type DWord -Force

    Write-Output "Suggested apps in Windows Ink Workspace have been disabled."
}

# Call the function
DisableAllowSuggestedAppsInWindowsInkWorkspace
