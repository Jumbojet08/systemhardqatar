# Ensure running as administrator
if (-NOT ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole] "Administrator")) {
    Write-Warning "Please run this script as Administrator!"
    exit
}

Write-Host "Starting password policy enforcement..." -ForegroundColor Cyan

# Method 1: Using net accounts command
try {
    Write-Host "Applying password policies using net accounts..." -ForegroundColor Cyan
    & net accounts /minpwlen:12
    & net accounts /maxpwage:60
    & net accounts /minpwage:1
    & net accounts /uniquepw:24
    & net accounts /lockoutthreshold:5
    & net accounts /lockoutduration:15
    & net accounts /lockoutwindow:15
    Write-Host "Net accounts policies applied successfully" -ForegroundColor Green
} catch {
    Write-Host "Error applying net accounts policies: $_" -ForegroundColor Red
}

# Method 2: Using Local Security Policy directly via secedit
try {
    Write-Host "Applying password complexity policy via secedit..." -ForegroundColor Cyan
    $tempFile = [System.IO.Path]::GetTempFileName() + ".inf" # Give it an .inf extension for clarity
    $dbPath = "$env:TEMP\custom_policy.sdb"

    @"
[Unicode]
Unicode=yes
[System Access]
PasswordComplexity = 1
[Version]
signature="`$CHICAGO`$"
Revision=1
"@ | Out-File -FilePath $tempFile -Encoding Unicode # <-- CHANGE HERE

    # Optional: Add logging for better diagnostics
    $logFile = "$env:TEMP\secedit_apply.log"

    Write-Host "Secedit command: secedit.exe /configure /db `"$dbPath`" /cfg `"$tempFile`" /areas SECURITYPOLICY /log `"$logFile`"" -ForegroundColor DarkGray

    # Execute secedit
    $result = Start-Process -FilePath "secedit.exe" -ArgumentList "/configure /db `"$dbPath`" /cfg `"$tempFile`" /areas SECURITYPOLICY /log `"$logFile`"" -NoNewWindow -Wait -PassThru

    if ($result.ExitCode -eq 0) {
        Write-Host "Password complexity policy applied successfully via secedit" -ForegroundColor Green
        if (Test-Path $logFile) {
             Write-Host "Secedit log available at: $logFile" -ForegroundColor DarkGray
        }
    } else {
        Write-Host "Failed to apply password complexity policy via secedit. Exit code: $($result.ExitCode)" -ForegroundColor Red
        # Display log content if it exists
        if (Test-Path $logFile) {
            Write-Warning "Secedit log content ($logFile):"
            Get-Content $logFile | Write-Warning
        } else {
            Write-Warning "Secedit log file not found."
        }
    }

    # Clean up
    Remove-Item -Path $tempFile -Force -ErrorAction SilentlyContinue
    Remove-Item -Path $dbPath -Force -ErrorAction SilentlyContinue
    # Remove-Item -Path $logFile -Force -ErrorAction SilentlyContinue # Keep log for debugging if needed

} catch {
    Write-Host "Error applying password complexity policy via secedit: $_" -ForegroundColor Red
    # Try to display log content on general error too
    if (Test-Path $logFile) {
        Write-Warning "Secedit log content ($logFile) during exception:"
        Get-Content $logFile | Write-Warning
    }
}
